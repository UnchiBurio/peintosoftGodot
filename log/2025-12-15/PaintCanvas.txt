script/source = "# PaintCanvasGrid.gd
extends Node2D

var grid_size: int = 50
var grid_color: Color = Color(0.2, 0.2, 0.2, 0.5)
var grid_visible: bool = true
var paint_canvas: Node2D

func _ready() -> void:
	paint_canvas = get_parent()
	queue_redraw()

func _draw() -> void:
	if not grid_visible:
		return
	
	var size = paint_canvas.canvas_size
	
	# 縦線の描画
	for x in range(0, int(size.x) + grid_size, grid_size):
		draw_line(
			Vector2(x, 0),
			Vector2(x, size.y),
			grid_color
		)
	
	# 横線の描画
	for y in range(0, int(size.y) + grid_size, grid_size):
		draw_line(
			Vector2(0, y),
			Vector2(size.x, y),
			grid_color
		)

func toggle_visibility() -> void:
	grid_visible = !grid_visible
	queue_redraw()
"
