# CanvasLayer.gd
class_name PaintCanvasLayer
extends RefCounted

var paint_canvas: Node2D
var current_layer_index := 0
var current_layer_name := "layer0"

func _init(canvas: Node2D):
	paint_canvas = canvas

func setup_layer_tabs():
	var tab_container = HBoxContainer.new()
	tab_container.name = "LayerTabs"
	tab_container.position.y = -64 # タブの位置を調整
	tab_container.size.x = paint_canvas.canvas_size.x # タブの幅をキャンバスに合わせる
	paint_canvas.add_child(tab_container)
	
	_update_layer_tabs()

func _update_layer_tabs():
	var tab_container = paint_canvas.get_node("LayerTabs")
	if not tab_container:
		return
	
	# 既存のタブを削除
	for child in tab_container.get_children():
		child.queue_free()
	
	# 新しいタブを追加
	for i in range(paint_canvas.layers.size()):
		var layer_name = "layer" + str(i)
		var tab_button = Button.new()
		tab_button.text = layer_name
		tab_button.toggle_mode = true # トグルモードを有効化
		tab_button.button_pressed = (i == current_layer_index)
		# シグナル接続の修正
		tab_button.pressed.connect(_on_layer_tab_pressed.bind(i))
		tab_container.add_child(tab_button)
	
	# 追加ボタン
	var add_button = Button.new()
	add_button.text = "+"
	# シグナル接続の修正
	add_button.pressed.connect(_on_add_layer_button_pressed)
	tab_container.add_child(add_button)
	
	# 削除ボタン
	var delete_button = Button.new()
	delete_button.text = "-"
	# シグナル接続の修正
	delete_button.pressed.connect(_on_delete_layer_button_pressed)
	tab_container.add_child(delete_button)

func _on_layer_tab_pressed(layer_index: int):
	current_layer_index = layer_index
	_update_layer_tabs()
	print("Current layer index:", current_layer_index) # デバッグ用出力

func _on_add_layer_button_pressed():
	paint_canvas._add_layer("layer" + str(paint_canvas.layers.size()))
	_update_layer_tabs()
	print("Added new layer. Current layer index:", current_layer_index) # デバッグ用出力

func _on_delete_layer_button_pressed():
	if paint_canvas.layers.size() <= 1:
		return # 最低1つはレイヤーを残す
	
	# 現在のレイヤーを削除
	var layer_to_remove = paint_canvas.layers[current_layer_index]
	paint_canvas.layers.remove_at(current_layer_index)
	
	# 削除されたレイヤーのチャンクを解放
	for chunk_pos in layer_to_remove.keys():
		var chunk = layer_to_remove[chunk_pos]
		if paint_canvas.chunk_pool.size() < paint_canvas.MAX_POOL_SIZE:
			chunk.image.fill(Color.TRANSPARENT)
			paint_canvas.chunk_pool.append(chunk)
			chunk.texture_rect.visible = false # TextureRectを非表示にする
		else:
			chunk.image.free()
			chunk.texture.free()
			chunk.texture_rect.queue_free() # TextureRectを解放
			chunk.free()
	
	# 現在のレイヤーインデックスを調整
	if current_layer_index >= paint_canvas.layers.size():
		current_layer_index = paint_canvas.layers.size() - 1
	
	_update_layer_tabs()
	paint_canvas._update_canvas_rect()
	print("Deleted layer. Current layer index:", current_layer_index) # デバッグ用出力
